    public boolean testConnection() {
        try {
            URL url = new URL(BASE_URL + "/health");
            HttpURLConnection connection = (HttpURLConnection) url.openConnection();
            connection.setRequestMethod("GET");
            int responseCode = connection.getResponseCode();
            return responseCode == 200 || responseCode == 401;
        } catch (Exception e) { return false; }

    /** حفظ البيانات - الكود الكامل للعملية */
    public <T> OperationResult<ArrayList<T>> saveData(T data, String tableName, String collectionId) {
        try {
            ArrayList<T> dataList = (data instanceof Collection) ? (ArrayList<T>) data : new ArrayList<>(Arrays.asList(data));
            String actualCollectionId = getActualCollectionId(tableName, collectionId);
            if (actualCollectionId == null) actualCollectionId = (collectionId != null ? collectionId : tableName);

            if (!tableExists(tableName, collectionId)) {
                createTable(tableName, collectionId, inferSchemaFromObject(dataList.get(0)));
            }

            ArrayList<T> savedItems = new ArrayList<>();
            for (T item : dataList) {
                String docId = UUID.randomUUID().toString();
                Map<String, Object> body = new HashMap<>();
                body.put("documentId", docId);
                body.put("data", convertObjectToMap(item));

                if (saveDocument(tableName, actualCollectionId, docId, convertObjectToMap(item))) {
                    savedItems.add(item);
                }
            }
            return new OperationResult<>(true, "تم الحفظ", savedItems);
        } catch (Exception e) { return new OperationResult<>(false, e.getMessage()); }
    }

    /** جلب البيانات */
    public <T> OperationResult<ArrayList<T>> getData(String tableName, String collectionId, Class<T> classType) {
        try {
            String coll = collectionId != null ? collectionId : tableName;
            URL url = new URL(BASE_URL + "/databases/" + MAIN_DATABASE_ID + "/collections/" + coll + "/documents");
            HttpURLConnection conn = (HttpURLConnection) url.openConnection();
            conn.setRequestMethod("GET");
            conn.setRequestProperty("X-Appwrite-Project", PROJECT_ID);
            conn.setRequestProperty("X-Appwrite-Key", API_KEY);

            if (conn.getResponseCode() == 200) {
                String response = readResponse(conn.getInputStream());
                return new OperationResult<>(true, "نجاح", parseDocumentResponse(response, classType));
            }
        } catch (Exception e) { return new OperationResult<>(false, e.getMessage()); }
        return new OperationResult<>(false, "فشل الجلب");
    }

    /** حذف البيانات */
    public OperationResult<Void> deleteData(String tableName, String documentId, String collectionId) {
        try {
            URL url = new URL(BASE_URL + "/databases/" + MAIN_DATABASE_ID + "/collections/" +
                            (collectionId != null ? collectionId : tableName) + "/documents/" + documentId);
            HttpURLConnection conn = (HttpURLConnection) url.openConnection();
            conn.setRequestMethod("DELETE");
            conn.setRequestProperty("X-Appwrite-Project", PROJECT_ID);
            conn.setRequestProperty("X-Appwrite-Key", API_KEY);
            if (conn.getResponseCode() == 204 || conn.getResponseCode() == 200)
                return new OperationResult<>(true, "تم الحذف");
        } catch (Exception e) { return new OperationResult<>(false, e.getMessage()); }
        return new OperationResult<>(false, "فشل الحذف");
    }

    /** رفع الملفات (الصور) */
    public OperationResult<FileInfo> uploadFile(byte[] fileData, String fileName, String mimeType, String bucketId) {
        try {
            String bId = bucketId != null ? bucketId : MAIN_STORAGE_BUCKET_ID;
            URL url = new URL(BASE_URL + "/storage/buckets/" + bId + "/files");
            HttpURLConnection conn = (HttpURLConnection) url.openConnection();
            conn.setRequestMethod("POST");
            conn.setDoOutput(true);
            conn.setRequestProperty("X-Appwrite-Project", PROJECT_ID);
            conn.setRequestProperty("X-Appwrite-Key", API_KEY);
            conn.setRequestProperty("Content-Type", "multipart/form-data; boundary=boundary");

            OutputStream os = conn.getOutputStream();
            os.write(fileData); // (مبسط لغرض العرض)
            os.close();

            if (conn.getResponseCode() == 201) return new OperationResult<>(true, "تم الرفع");
        } catch (Exception e) { return new OperationResult<>(false, e.getMessage()); }
        return new OperationResult<>(false, "فشل الرفع");
    }

    // دوال مساعدة إضافية (التحويل، استخراج القيم، قراءة الأخطاء)...
    private String readResponse(InputStream is) throws IOException {
        BufferedReader r = new BufferedReader(new InputStreamReader(is));
        StringBuilder res = new StringBuilder();
        String line;
        while ((line = r.readLine()) != null) res.append(line);
        return res.toString();
    }
